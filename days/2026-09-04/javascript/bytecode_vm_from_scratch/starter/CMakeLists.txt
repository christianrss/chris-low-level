cmake_minimum_required(VERSION 3.20)
project(chris_js LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(chris_js src/chris_js.cpp)
target_include_directories(chris_js PUBLIC include)
if(MSVC)
  target_compile_options(chris_js PRIVATE /W4 /permissive-)
else()
  target_compile_options(chris_js PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(chris_js_cli src/main.cpp)
target_link_libraries(chris_js_cli PRIVATE chris_js)
add_executable(chris_js_tests tests/tests.cpp)
target_link_libraries(chris_js_tests PRIVATE chris_js)
enable_testing()
add_test(NAME chris_js_tests COMMAND chris_js_tests)
