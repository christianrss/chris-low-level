cmake_minimum_required(VERSION 3.20)
project(chris_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(chris_cpu src/cpu.cpp)
target_include_directories(chris_cpu PUBLIC include)
if(MSVC)
  target_compile_options(chris_cpu PRIVATE /W4 /permissive-)
else()
  target_compile_options(chris_cpu PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(cpu_tests tests/test_cpu.cpp)
target_link_libraries(cpu_tests PRIVATE chris_cpu)
option(CHRIS_BUILD_BENCHMARKS "Build benchmark targets" OFF)
if(CHRIS_BUILD_BENCHMARKS)
  add_executable(cpu_benchmark benchmarks/benchmark.cpp)
  target_link_libraries(cpu_benchmark PRIVATE chris_cpu)
endif()
enable_testing()
add_test(NAME cpu_tests COMMAND cpu_tests)
