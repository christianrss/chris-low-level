cmake_minimum_required(VERSION 3.20)
project(chris_spsc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(chris_spsc src/spsc.cpp)
target_include_directories(chris_spsc PUBLIC include)
add_executable(chris_spsc_tests tests/test_spsc.cpp)
target_link_libraries(chris_spsc_tests PRIVATE chris_spsc)
enable_testing()
add_test(NAME chris_spsc_tests COMMAND chris_spsc_tests)
