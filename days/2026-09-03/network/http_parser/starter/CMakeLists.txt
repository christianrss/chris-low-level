cmake_minimum_required(VERSION 3.20)
project(chris_http LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(chris_http src/http_parser.cpp)
target_include_directories(chris_http PUBLIC include)
if(MSVC)
  target_compile_options(chris_http PRIVATE /W4 /permissive-)
else()
  target_compile_options(chris_http PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(http_tests tests/test_http.cpp)
target_link_libraries(http_tests PRIVATE chris_http)
option(CHRIS_BUILD_BENCHMARKS "Build benchmark targets" OFF)
if(CHRIS_BUILD_BENCHMARKS)
  add_executable(http_benchmark benchmarks/benchmark.cpp)
  target_link_libraries(http_benchmark PRIVATE chris_http)
endif()
enable_testing()
add_test(NAME http_tests COMMAND http_tests)
