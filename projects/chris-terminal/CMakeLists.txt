cmake_minimum_required(VERSION 3.20)
project(chris_terminal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(chris_terminal src/terminal.cpp)
target_include_directories(chris_terminal PUBLIC include)
if(MSVC)
  target_compile_options(chris_terminal PRIVATE /W4 /permissive-)
else()
  target_compile_options(chris_terminal PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(terminal_tests tests/test_terminal.cpp)
target_link_libraries(terminal_tests PRIVATE chris_terminal)
option(CHRIS_BUILD_BENCHMARKS "Build benchmark targets" OFF)
if(CHRIS_BUILD_BENCHMARKS)
  add_executable(terminal_benchmark benchmarks/benchmark.cpp)
  target_link_libraries(terminal_benchmark PRIVATE chris_terminal)
endif()
enable_testing()
add_test(NAME terminal_tests COMMAND terminal_tests)
