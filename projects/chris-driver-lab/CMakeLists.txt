cmake_minimum_required(VERSION 3.20)
project(chris_driver_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(descriptor_ring src/descriptor_ring.cpp)
target_include_directories(descriptor_ring PUBLIC include)
if(MSVC)
  target_compile_options(descriptor_ring PRIVATE /W4 /permissive-)
else()
  target_compile_options(descriptor_ring PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(driver_ring_tests tests/test_ring.cpp)
target_link_libraries(driver_ring_tests PRIVATE descriptor_ring)
option(CHRIS_BUILD_BENCHMARKS "Build benchmark targets" OFF)
if(CHRIS_BUILD_BENCHMARKS)
  add_executable(driver_ring_benchmark benchmarks/benchmark.cpp)
  target_link_libraries(driver_ring_benchmark PRIVATE descriptor_ring)
endif()
enable_testing()
add_test(NAME driver_ring_tests COMMAND driver_ring_tests)
