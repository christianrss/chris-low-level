cmake_minimum_required(VERSION 3.20)
project(chris_algorithms LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(chris_algorithms src/blocked_sort.cpp)
target_include_directories(chris_algorithms PUBLIC include)
if(MSVC)
  target_compile_options(chris_algorithms PRIVATE /W4 /permissive-)
else()
  target_compile_options(chris_algorithms PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(algorithm_tests tests/algorithm_tests.cpp)
target_link_libraries(algorithm_tests PRIVATE chris_algorithms)

option(CHRIS_BUILD_BENCHMARKS "Build benchmark targets" OFF)
if(CHRIS_BUILD_BENCHMARKS)
  add_executable(algorithm_benchmark benchmarks/algorithm_benchmark.cpp)
  target_link_libraries(algorithm_benchmark PRIVATE chris_algorithms)
endif()

enable_testing()
add_test(NAME algorithm_tests COMMAND algorithm_tests)
