cmake_minimum_required(VERSION 3.20)
project(chris_assembly_lab LANGUAGES C)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  enable_language(ASM)
  add_library(asm_sum src/sum_x86_64.S)
  add_executable(assembly_tests tests/test_sum.c)
  target_link_libraries(assembly_tests PRIVATE asm_sum)
  option(CHRIS_BUILD_BENCHMARKS "Build benchmark targets" OFF)
  if(CHRIS_BUILD_BENCHMARKS)
    add_executable(assembly_benchmark benchmarks/benchmark.c)
    target_link_libraries(assembly_benchmark PRIVATE asm_sum)
  endif()
  enable_testing()
  add_test(NAME assembly_tests COMMAND assembly_tests)
else()
  message(STATUS "chris-assembly-lab Day 01 executable target is Linux x86-64 System V only; project is skipped on this platform.")
  enable_testing()
endif()
