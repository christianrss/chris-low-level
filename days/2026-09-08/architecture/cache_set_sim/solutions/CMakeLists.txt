cmake_minimum_required(VERSION 3.20)
project(chris_cache LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(chris_cache src/cache.cpp)
target_include_directories(chris_cache PUBLIC include)
add_executable(chris_cache_tests tests/test_cache.cpp)
target_link_libraries(chris_cache_tests PRIVATE chris_cache)
enable_testing()
add_test(NAME chris_cache_tests COMMAND chris_cache_tests)
